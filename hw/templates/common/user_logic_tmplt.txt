{% endif %}

    AXI4L.s                     s_axi_coprocessor_mmio [N_COPROCESSOR_PORTS],
    input coprocessor_port_status_t coprocessor_status [N_COPROCESSOR_PORTS],

{% endif %}
{% if cnfg.en_mem %}
    // CARD DATA STREAMS
    AXI4SR.s                    axis_card_recv [N_CARD_AXI],
    AXI4SR.m                    axis_card_send [N_CARD_AXI],

{% endif %}
{% if cnfg.en_rdma %}
    // RDMA DATA STREAMS REQUESTER
    AXI4SR.s                    axis_rreq_recv [N_RDMA_AXI],
    AXI4SR.m                    axis_rreq_send [N_RDMA_AXI],

    // RDMA DATA STREAMS RESPONDER
    AXI4SR.s                    axis_rrsp_recv [N_RDMA_AXI],
    AXI4SR.m                    axis_rrsp_send [N_RDMA_AXI],

{% endif %}
{% if cnfg.en_tcp %}
    // TCP/IP DATA STREAMS
    AXI4SR.s                    axis_tcp_recv [N_TCP_AXI],
    AXI4SR.m                    axis_tcp_send [N_TCP_AXI],

{% endif %}

{% if ( cnfg.en_sniffer and ( cnfg.sniffer_vfpga_id == c_reg ) ) %}
    AXI4S.s                     axis_rx_sniffer,
    AXI4S.s                     axis_tx_sniffer,
    metaIntf.m                  filter_config,
{% endif %}

{% if ( cnfg.en_net_raw and ( cnfg.net_raw_vfpga_id == c_reg ) ) %}
    // Raw Ethernet: whole frames, 512-bit, aclk domain. This vFPGA is the MAC
    // client - it must build and parse the full frame itself, including
    // destination MAC, source MAC and EtherType. mac_status is
    // {gt_rst_done_rx, gt_rst_done_tx, rst_done, rst_state[4:0]}.
    AXI4S.s                     axis_net_raw_rx,
    AXI4S.m                     axis_net_raw_tx,
    input  wire[7:0]            mac_status,
    // Debug register port into the MAC (aclk). Pulse mac_reg_req with
    // {mac_reg_wr, mac_reg_addr, mac_reg_wdata} valid, then wait for the
    // mac_reg_rsp pulse before issuing another. On Versal, addresses below
    // 0x100000 are the DCMAC's PG369 registers and 0x100000 up is the segment
    // monitor in network/dcmac_cyt/dcmac_debug.sv. Tie mac_reg_req to 0 if
    // unused.
    output logic                mac_reg_req,
    output logic                mac_reg_wr,
    output logic[31:0]          mac_reg_addr,
    output logic[31:0]          mac_reg_wdata,
    input  wire                 mac_reg_rsp,
    input  wire[31:0]           mac_reg_rdata,
    input  wire[1:0]            mac_reg_resp,
{% endif %}

{% if (cnfg.en_aurora_1 or cnfg.en_qlink_gty) and c_reg == 0 %}
    AXI4S.s                     axis_aurora_rx,
    AXI4S.m                     axis_aurora_tx,
    input  wire                 aurora_channel_up,
    input  wire[3:0]            aurora_lane_up,
{% endif %}

{% if cnfg.en_qlink_gty and c_reg == 0 %}
    // QLINK: axis_aurora_tx is in qlink_tx_clk, axis_aurora_rx in qlink_rx_clk.
    input  wire                 qlink_tx_clk,
    input  wire                 qlink_rx_clk,
    input  wire                 qlink_tx_rstn,
    input  wire                 qlink_rx_rstn,
    output wire[2:0]            qlink_loopback_sel,
{% endif %}

{% if cnfg.en_amc and c_reg == 0 %}
    // On-card R5<->vFPGA OCM mailbox (Example 15). A raw 32-bit AXI4 master out
    // of the vFPGA that the shell routes to the CIPS S_AXI_LPD -> OCM, plus a
    // doorbell interrupt to the R5 (pl_ps_irq1). AMC only, region 0.
    AXI4.m                      m_axi_ocm,
    output logic                ocm_irq,
{% endif %}

    // Clock and reset
    input  wire                 aclk,
    input  wire[0:0]            aresetn
);
