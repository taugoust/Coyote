`timescale 1ns / 1ps

import lynxTypes::*;

// Keeps the simulation hierarchy stable while selecting the registered module
// at generation time. The ports are the external dynamic-service contract.
module design_dynamic_service_sim (
    AXI4S.s                     s_axis_host_in [N_REGIONS],
    AXI4S.m                     m_axis_host_in [N_REGIONS],
    AXI4S.s                     s_axis_host_out [N_REGIONS],
    AXI4S.m                     m_axis_host_out [N_REGIONS],
    AXI4L.s                     s_axi_ctrl,
{% if cnfg.en_external_dynamic_service_peer_endpoints %}
    AXI4SR.s                    s_axis_peer_recv [N_PEER_AXI],
    AXI4SR.m                    m_axis_peer_send [N_PEER_AXI],
    input logic [N_PEER_LINKS-1:0] peer_link_up,
    input logic [(4*N_PEER_LINKS)-1:0] peer_lane_up,
{% endif %}
{% if cnfg.en_external_dynamic_service_slot_status %}
    input logic [N_REGIONS-1:0] s_slot_decoupled,
{% endif %}
    input logic                 aclk,
    input logic                 aresetn
);

    {{ cnfg.external_dynamic_service_top }} inst_external_dynamic_service (
{% if cnfg.en_external_dynamic_service_control %}
        .s_axi_ctrl(s_axi_ctrl),
{% endif %}
{% if cnfg.en_external_dynamic_service_slot_status %}
        .s_slot_decoupled(s_slot_decoupled),
{% endif %}
{% if cnfg.en_external_dynamic_service_peer_endpoints %}
        .s_axis_peer_recv(s_axis_peer_recv),
        .m_axis_peer_send(m_axis_peer_send),
        .peer_link_up(peer_link_up),
        .peer_lane_up(peer_lane_up),
{% endif %}
        .aclk(aclk),
        .aresetn(aresetn),
        .s_axis_host_in(s_axis_host_in),
        .m_axis_host_in(m_axis_host_in),
        .s_axis_host_out(s_axis_host_out),
        .m_axis_host_out(m_axis_host_out)
    );

{% if not cnfg.en_external_dynamic_service_control %}
    always_comb s_axi_ctrl.tie_off_s();
{% endif %}

endmodule
