cmake_minimum_required(VERSION 3.15)
project(coyote-application-reset-configuration LANGUAGES NONE)
if(NOT DEFINED CYT_DIR)
    message(FATAL_ERROR "CYT_DIR must name Coyote source")
endif()
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}" "${CYT_DIR}/cmake")
find_package(CoyoteHW REQUIRED)
set(EN_STRM 1)
set(N_STRM_AXI 1)
set(EN_MEM 0)
set(EN_RDMA 0)
set(EN_TCP 0)
set(N_CONFIG 1)
set(EN_PR 1)
set(STATIC_PATH "${CMAKE_CURRENT_SOURCE_DIR}/static")
set(FPLAN_PATH "${CMAKE_CURRENT_SOURCE_DIR}/fixture_floorplan.xdc")
if(FDEV_NAME STREQUAL "v80")
    set(EN_SHELL_PBLOCK 0)
else()
    set(EN_SHELL_PBLOCK 1)
endif()
set(TEST_N_REGIONS 1 CACHE STRING "Region count")
set(N_REGIONS ${TEST_N_REGIONS})
math(EXPR LAST_REGION "${N_REGIONS} - 1")
foreach(REGION RANGE ${LAST_REGION})
    set(vfpga_c0_${REGION} "${CMAKE_CURRENT_SOURCE_DIR}/dummy_app")
endforeach()
validation_checks_hw()
set(LOAD_APPS 0)
gen_scripts()
